from collections import deque
start = (0, 0)
goal = (2, 2)
obstacles = [(1, 1)]
moves = [(0, 1), (1, 0), (0, -1), (-1, 0)]

queue = deque([(start, [start])])
visited = {start}

while queue:

    pos, path = queue.popleft()

    print("Robot at:", pos)

    if pos == goal:
        print("\nGoal Reached!")
        print("Path:", path)
        print("Total Steps:", len(path) - 1)
        break

    for dx, dy in moves:

        x = pos[0] + dx
        y = pos[1] + dy

        new_pos = (x, y)

        if (0 <= x < 3 and 0 <= y < 3
            and new_pos not in obstacles
            and new_pos not in visited):

            visited.add(new_pos)
            queue.append((new_pos, path + [new_pos]))

else:
    print("No Path Found")
